# Install this file separately, with --no-deps, AFTER requirements.txt:
#   pip install --no-deps -r requirements-face-recognition.txt
#
# Why: face-recognition's packaged metadata declares a dependency on
# "dlib", but on Windows there is no prebuilt cp311 wheel for canonical
# dlib, so a normal install would try to compile it from source (requiring
# Visual C++ Build Tools). We use "dlib-bin" instead, which provides the
# same importable `dlib` module as a prebuilt wheel. Installing
# face-recognition with --no-deps prevents pip from trying to fetch/build
# canonical dlib.
#
# This causes "pip check" to report:
#   face-recognition 1.3.0 requires dlib, which is not installed.
# This is expected and harmless — dlib-bin satisfies the actual import at
# runtime. Do not attempt to "fix" this by installing canonical dlib.
face-recognition==1.3.0
